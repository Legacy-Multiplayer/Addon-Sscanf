cmake_minimum_required(VERSION 3.20)
project(AddonSscanf VERSION 1.0.0 LANGUAGES CXX)

option(SSCANF_BUILD_TESTS "Build the scanner unit tests" ON)

if(NOT EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/SDK/CMakeLists.txt")
    message(FATAL_ERROR "Initialize Addon-SDK: git submodule update --init --recursive")
endif()
add_subdirectory(SDK)
include(SDK/cmake/Plugin.cmake)

add_library(ScannerCore STATIC src/Scanner.cpp)
target_compile_features(ScannerCore PUBLIC cxx_std_17)
target_include_directories(ScannerCore PUBLIC src)
set_target_properties(ScannerCore PROPERTIES POSITION_INDEPENDENT_CODE ON CXX_EXTENSIONS OFF)
if(MSVC)
    target_compile_options(ScannerCore PRIVATE /W4 /utf-8 /permissive-)
else()
    target_compile_options(ScannerCore PRIVATE -Wall -Wextra)
endif()

legacy_plugin(Addon-Sscanf src/Main.cpp)
target_link_libraries(Addon-Sscanf PRIVATE ScannerCore)

if(SSCANF_BUILD_TESTS)
    enable_testing()
    add_executable(LoadTests tests/LoadTests.cpp)
    target_link_libraries(LoadTests PRIVATE Legacy::SDK ${CMAKE_DL_LIBS})
    add_dependencies(LoadTests Addon-Sscanf)
    add_test(NAME PluginExports COMMAND LoadTests $<TARGET_FILE:Addon-Sscanf>)
    add_executable(ScannerTests tests/ScannerTests.cpp)
    target_link_libraries(ScannerTests PRIVATE ScannerCore)
    add_test(NAME ScannerTests COMMAND ScannerTests)
endif()
